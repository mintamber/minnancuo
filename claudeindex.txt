      const timeline = [["Migration and Settlement","迁徙与落地","Northern traditions met the local mountain-sea environment, forming settlements with both ritual order and regional character."],["Lineage and Main Hall","家族与厅堂","Axis, forecourt, hall, and ancestral space made the house a core setting for family relations and ceremonial life."],["Maritime Links","海洋与侨乡","Trade routes and overseas remittances enriched decoration, materials, and taste."],["Contemporary Value","当代再认识","Today, Cuò dwellings are cultural heritage and a lens for local identity, craft memory, and community continuity."]];
      const features = [["Red Brick and Pale Stone","Warm red brick walls contrast with granite bases, creating a weather-resistant and grounded image."],["Courtyard Life","Forecourts, inner yards, and halls support rituals, gatherings, drying, seasonal work, and everyday conversation."],["Curved Ridges","Upturned roof ends and flowing ridgelines give the building a light, animated profile."],["Crafted Surfaces","Wall panels, windows, and ridges often carry flowers, auspicious beasts, and geometric patterns."],["Climate Response","Raised stone bases, thick walls, and pitched roofs answer humidity, rain, and coastal winds."],["Ritual Order","Gate, courtyard, hall, and side wings create a sequence between outside and inside, guest and family, elder and younger."]];
      const recap = [["Material","Brick, stone, plaster"],["Space","Court, hall, wings"],["Roof","Pitch, curve, swallowtail"],["Ornament","Flowers, beasts, lattice"],["Elements","Point, rise, round, wave, flat"]];
      const gables = [{id:"jin",name:"Metal",zh:"金",image:ASSETS.jixing,cue:"Metal (金) gables often feel rounded, contained, and bell-like. Look for a composed crown and a stable central rise.",wrongCue:"Look for the rounded, contained crown rather than a sharp flame or flowing wave."},{id:"mu",name:"Wood",zh:"木",image:ASSETS.muxing,cue:"Wood (木) gables rise upward with a strong vertical center, like a trunk or growing branch.",wrongCue:"Look for the strong upward movement and vertical central spine."},{id:"shui",name:"Water",zh:"水",image:ASSETS.shuixing,cue:"Water (水) gables have a soft, wave-like top. Their outline flows horizontally with repeated rises and falls.",wrongCue:"Look for a continuous wavy silhouette rather than a single high point."},{id:"huo",name:"Fire",zh:"火",image:ASSETS.huoxing,cue:"Fire (火) gables push upward into a pointed peak. The sides gather toward the top like an ascending flame.",wrongCue:"Look for the sharp central peak and flame-like upward energy."},{id:"tu",name:"Earth",zh:"土",image:ASSETS.tuxing,cue:"Earth (土) gables are broad, grounded, and comparatively flat.",wrongCue:"Look for a broad, steady, grounded outline with less dramatic height change."}];
      const $ = s => document.querySelector(s); const $$ = s => [...document.querySelectorAll(s)];
      let quizIndex=0, correctCount=0, wrongAttempts=0, resolved=false, userPausedMusic=false;

      function renderTimeline(){ $('#historyTimeline').innerHTML = timeline.map((x,i)=>`<article class="timeline-card" data-step="0${i+1}" data-tilt><h3>${x[0]}</h3><span class="zh-subtitle">${x[1]}</span><p>${x[2]}</p></article>`).join(''); }
      function renderFeatures(){ $('#featureGrid').innerHTML = features.map((x,i)=>`<article class="card" data-tilt><span class="number">FEATURE 0${i+1}</span><h3>${x[0]}</h3><p>${x[1]}</p></article>`).join(''); }
      function renderRecap(){ $('#recapStrip').innerHTML = recap.map(x=>`<article class="recap-item" data-tilt><strong>${x[0]}</strong><span>${x[1]}</span></article>`).join(''); }

      const AUDIO_PATHS = {
        bgm: ["./bgm.mp3", "./image/bgm.mp3", "./audio/bgm.mp3", "bgm.mp3"],
        correctSound: ["./correct.mp3", "./image/correct.mp3", "./audio/correct.mp3", "correct.mp3"],
        trySound: ["./tryagain.mp3", "./image/tryagain.mp3", "./audio/tryagain.mp3", "tryagain.mp3"]
      };
      function setAudioSource(id, index = 0) {
        const audio = document.getElementById(id);
        const paths = AUDIO_PATHS[id] || [];
        if (!audio || !paths[index]) return;
        if (!audio.src.endsWith(paths[index])) audio.src = paths[index];
        audio.dataset.sourceIndex = String(index);
        audio.onerror = () => setAudioSource(id, index + 1);
        audio.load();
      }
      function playSound(id) {
        const audio = document.getElementById(id);
        if (!audio) return;
        const index = Number(audio.dataset.sourceIndex || 0);
        const tryCurrent = () => { audio.currentTime = 0; return audio.play(); };
        tryCurrent().catch(() => {
          const paths = AUDIO_PATHS[id] || [];
          const next = index + 1;
          if (paths[next]) { setAudioSource(id, next); setTimeout(() => audio.play().catch(() => {}), 80); }
        });
      }
      function updateProgress(){ $('#progressBar').style.width = `${(quizIndex/gables.length)*100}%`; }
      function renderQuiz(){
        updateProgress();
        if(quizIndex>=gables.length) return renderResult();
        const c=gables[quizIndex]; wrongAttempts=0; resolved=false;
        $('#gameBoard').innerHTML=`<figure class="image-card contain quiz-image" data-tilt><img src="${c.image}" alt="${c.name} gable" /></figure><article class="quiz-panel"><div class="quiz-meta"><span>Question ${quizIndex+1}/${gables.length}</span><span>Correct ${correctCount}/${gables.length}</span></div><h3>Which Five-Element Gable is this? <span class="title-zh">（五行山墙）</span></h3><p>Choose by silhouette first: point, rise, roundness, wave, or grounded flatness.</p><div class="choices">${gables.map(g=>`<button class="choice-btn" type="button" data-answer="${g.id}">${g.name}</button>`).join('')}</div><div class="feedback" id="feedback"></div></article>`;

        $$('.choice-btn').forEach(btn=>{
          btn.addEventListener('click',(e)=>{ addRipple(e, btn); checkAnswer(btn,c); });
        });
        attachTilt();
      }
      function resolveQuestion(ok,c){
        resolved=true;

        $$('.choice-btn').forEach(ch=>{ch.disabled=true; if(ch.dataset.answer===c.id) ch.classList.add('correct');});
        $('#feedback').innerHTML=`<div class="learning-card"><strong>${ok?'✓ Correct.':'◇ Answer revealed.'}</strong><br>${c.name}<span class="title-zh">${c.zh}</span>${c.cue}</div><button class="next-btn" type="button" id="nextQuestion">${quizIndex===gables.length-1?'See Result':'Next Question'}</button>`;
        $('#nextQuestion').addEventListener('click',(e)=>{ addRipple(e, e.currentTarget); setTimeout(()=>{ quizIndex++; renderQuiz(); }, 180); });
      }
      function checkAnswer(btn,c){
        if(resolved) return;
        const ok=btn.dataset.answer===c.id;
        if(ok){correctCount++; playSound('correctSound'); btn.classList.add('correct'); resolveQuestion(true,c); return;}
        wrongAttempts++; btn.classList.add('wrong'); btn.disabled=true;
        if(wrongAttempts===1){ playSound('trySound'); $('#feedback').innerHTML=`<div class="learning-card"><strong>↻ Try again.</strong><br>${c.wrongCue}</div>`; return; }
        resolveQuestion(false,c);
      }
      function renderResult(){
        $('#progressBar').style.width='100%';
        $('#gameBoard').innerHTML=`<figure class="image-card contain quiz-image" data-tilt><img src="${ASSETS.overview}" alt="Minnan dwelling overview" /></figure><article class="quiz-panel"><div class="quiz-meta"><span>Quiz Complete</span><span>Correct ${correctCount}/${gables.length}</span></div><h3>Your Five-Element Gable score is ${correctCount}/${gables.length}. <span class="title-zh">（五行山墙）</span></h3><p>Fire points upward, Wood rises vertically, Water flows in waves, Metal rounds and gathers, and Earth stays broad and grounded.</p><button class="restart-btn" type="button" id="restartQuiz">Restart Quiz</button></article>`;
        $('#restartQuiz').addEventListener('click',(e)=>{ addRipple(e, e.currentTarget); setTimeout(()=>{ quizIndex=0; correctCount=0; renderQuiz(); }, 180); });
        attachTilt();
      }
      function setupMusic(){
        setAudioSource('bgm'); setAudioSource('correctSound'); setAudioSource('trySound');
        const bgm=$('#bgm'), toggle=$('#musicToggle');
        const setPaused=()=>toggle.classList.toggle('is-paused', bgm.paused);
        bgm.volume=.35;
        const tryPlay=()=>{if(userPausedMusic) return; bgm.play().then(setPaused).catch(setPaused);};
        tryPlay();
        window.addEventListener('load',tryPlay,{once:true});
        document.addEventListener('pointerdown',tryPlay,{once:true});
        document.addEventListener('keydown',tryPlay,{once:true});
        toggle.addEventListener('click',()=>{ if(bgm.paused){ userPausedMusic=false; bgm.play().catch(()=>{}); } else { userPausedMusic=true; bgm.pause(); } setTimeout(setPaused,80); });
        bgm.addEventListener('play',setPaused);
        bgm.addEventListener('pause',setPaused);
      }

      /* ===== 涟漪点击效果 ===== */
      function addRipple(e, el) {
        const rect = el.getBoundingClientRect();
        const ripple = document.createElement('span');
        ripple.className = 'ripple';
        const size = Math.max(rect.width, rect.height);
        ripple.style.width = ripple.style.height = size + 'px';
        ripple.style.left = (e.clientX - rect.left - size/2) + 'px';
        ripple.style.top = (e.clientY - rect.top - size/2) + 'px';
        el.appendChild(ripple);
        setTimeout(()=>ripple.remove(), 700);
      }
      document.addEventListener('click', (e) => {
        const target = e.target.closest('.primary-link, .ghost-link');
        if (target && !target.querySelector('.ripple')) addRipple(e, target);
      });

      /* ===== 鼠标光晕跟随 ===== */
      const cursorGlow = $('#cursorGlow');
      let glowTargetX = 0, glowTargetY = 0, glowX = 0, glowY = 0;
      window.addEventListener('pointermove', (e) => {
        glowTargetX = e.clientX; glowTargetY = e.clientY;
        cursorGlow.style.opacity = '1';
      });
      window.addEventListener('pointerleave', () => { cursorGlow.style.opacity = '0'; });
      function animateGlow() {
        glowX += (glowTargetX - glowX) * 0.12;
        glowY += (glowTargetY - glowY) * 0.12;
        cursorGlow.style.transform = `translate(${glowX}px, ${glowY}px) translate(-50%, -50%)`;
        requestAnimationFrame(animateGlow);
      }
      animateGlow();

      /* ===== 卡片3D倾斜 ===== */
      function attachTilt() {

        $$('[data-tilt]').forEach(el => {
          if (el.dataset.tiltBound) return;
          el.dataset.tiltBound = '1';
          el.style.transformStyle = 'preserve-3d';
          el.style.willChange = 'transform';
          el.addEventListener('pointermove', (e) => {
            const rect = el.getBoundingClientRect();
            const px = (e.clientX - rect.left) / rect.width - 0.5;
            const py = (e.clientY - rect.top) / rect.height - 0.5;
            const rx = (-py * 6).toFixed(2);
            const ry = (px * 6).toFixed(2);
            el.style.transform = `perspective(1000px) rotateX(${rx}deg) rotateY(${ry}deg) translateY(-6px)`;
          });
          el.addEventListener('pointerleave', () => { el.style.transform = ''; });
        });
      }

      /* ===== 飘落花瓣（梅花/茉莉造型） ===== */
      function spawnPetals(count = 14) {
        const layer = $('#petalsLayer');
        const colors = ['#c8442a', '#b58a3b', '#d9a06a', '#e4c068', '#a04a30'];
        const svgFlower = (color) => `<svg viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><g fill="${color}" opacity="0.75"><circle cx="12" cy="6" r="4"/><circle cx="18" cy="12" r="4"/><circle cx="12" cy="18" r="4"/><circle cx="6" cy="12" r="4"/><circle cx="12" cy="12" r="3" fill="#e4c068"/></g></svg>`;
        for (let i = 0; i < count; i++) {
          const p = document.createElement('div');
          p.className = 'petal';
          const size = 10 + Math.random() * 14;
          p.style.width = p.style.height = size + 'px';
          p.style.left = (Math.random() * 100) + 'vw';
          p.style.animationDuration = (14 + Math.random() * 16) + 's';
          p.style.animationDelay = (Math.random() * 20) + 's';
          p.innerHTML = svgFlower(colors[i % colors.length]);
          layer.appendChild(p);
        }
      }

      /* ===== 滚动渐入 ===== */
      const observer = new IntersectionObserver(entries=>entries.forEach(e=>{ if(e.isIntersecting) e.target.classList.add('is-visible'); }), {threshold:.12});

      $$('.section-reveal').forEach(s=>observer.observe(s));

      renderTimeline(); renderFeatures(); renderRecap(); renderQuiz(); setupMusic();
      attachTilt();
      spawnPetals(16);
    </script>
  </body>
